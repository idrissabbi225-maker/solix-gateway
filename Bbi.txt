    <script>
        let currentCurrency = 'SLX';
        let currentFee = 0;
        const baseAmount = 45.00;

        document.getElementById('supplierSelect').addEventListener('change', function(e) {
            const val = e.target.value;
            let name = 'CJ Dropshipping';
            if(val === 'ALIEXPRESS') name = 'AliExpress API';
            if(val === 'ZENDROP') name = 'Zendrop Fulfillment';
            if(val === 'PRIVATE') name = 'Fournisseur Privé/Local';
            document.getElementById('supplierStatusText').innerText = 'Synchro active : ' + name;
        });

        function selectCurrency(currency, fee) {
            currentCurrency = currency;
            currentFee = fee;

            document.querySelectorAll('.currency-card').forEach(card => {
                card.classList.remove('border-blue-600', 'bg-blue-50/40');
                card.classList.add('border-slate-200', 'bg-white');
            });

            const activeCard = document.getElementById('curr-' + currency);
            if(activeCard) {
                activeCard.classList.remove('border-slate-200', 'bg-white');
                activeCard.classList.add('border-blue-600', 'bg-blue-50/40');
            }

            document.getElementById('feePercentageLabel').innerText = fee;
            
            if(fee === 0) {
                document.getElementById('feeAmountLabel').innerHTML = '0.00 $ (Économie 100%)';
                document.getElementById('totalAmountLabel').innerText = baseAmount.toFixed(2) + ' ' + currency;
            } else {
                const feeVal = (baseAmount * fee) / 100;
                const totalVal = baseAmount + feeVal;
                document.getElementById('feeAmountLabel').innerText = feeVal.toFixed(2) + ' $';
                document.getElementById('totalAmountLabel').innerText = totalVal.toFixed(2) + ' ' + currency;
            }
        }

        function openWalletModal() {
            document.getElementById('walletModal').style.display = 'flex';
        }

        function closeWalletModal() {
            document.getElementById('walletModal').style.display = 'none';
        }

        function connectWallet(walletName) {
            closeWalletModal();
            const btn = document.getElementById('connectBtn');
            btn.innerHTML = `<span class="w-2 h-2 rounded-full bg-emerald-400"></span><span>0x71C...49a</span>`;
            
            const walletInfo = document.getElementById('walletInfo');
            walletInfo.innerHTML = `
                <p class="font-bold text-slate-800">Connecté via ${walletName}</p>
                <p class="text-slate-600 font-mono text-[11px]">0x71C839...E49a</p>
                <div class="mt-3 pt-2 border-t border-slate-100 grid grid-cols-2 gap-2 text-xs">
                    <div><strong>Solde SLX :</strong> 1,250.00</div>
                    <div><strong>Solde USDT :</strong> 340.50</div>
                </div>
            `;
        }

        function executeEscrowPayment() {
            const supplier = document.getElementById('supplierSelect').value;
            const statusBox = document.getElementById('escrowStatusBox');
            statusBox.classList.remove('hidden');
            
            document.getElementById('routingSuccessText').innerText = `Fonds sécurisés dans l'escrow. Ordre routé avec succès vers l'API de ${supplier}.`;

            const payBtn = document.getElementById('payBtn');
            payBtn.innerText = "Fonds sous séquestre (Routage actif)";
            payBtn.disabled = true;
            payBtn.classList.add('opacity-50', 'cursor-not-allowed');
        }
    </script>
